// ls cmd

// current directory
ls

// directory
ls /home/user
ls /home/user/docs

// file
ls /home/user/notes.txt

// multiple paths
ls /home/user /tmp
ls /home/user/notes.txt /etc

// errors
ls /does-not-exist
ls /home/user /does-not-exist

cd /home/user/docs

cd docs

cd ~
pwd

cd /
pwd

cd /tmp

pwd

// no arguments
cd
pwd

cd /does-not-exist
